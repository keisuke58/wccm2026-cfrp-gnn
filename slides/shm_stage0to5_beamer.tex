\begin{frame}{Motivation: reusable rockets change the question}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{Expendable:} ``will it survive \emph{one} flight?'' --- a single
pass/fail margin check.\\[1.5mm]
\textbf{Reusable:} \alert{``is this vehicle OK to fly \emph{again}, and how many
flights remain?''} --- a recurring, history-dependent decision.
\begin{itemize}
  \item Same vehicle re-inspected after every flight
        $\Rightarrow$ \emph{fixed geometry, fixed individual} is the norm,
        and its own past-flight scans form a growing healthy reference
  \item SpaceX (empirically): preventive replacement $\to$
        \emph{condition-based monitoring}; staged certification
        $20\to40$ flights; fleet-leader B1067 at 35 flights (06/2026)
  \item Cost asymmetry: vehicle + payload loss $\gg$ unnecessary refurbishment
\end{itemize}
\column{0.42\textwidth}
\begin{block}{This work}
Formalise that practice as a \emph{measurable, Bayesian, physics-based}
pipeline --- detect $\to$ characterise (with uncertainty) $\to$ predict crack
growth $\to$ clear --- applicable from the \emph{first} article, not after a
decade of flights.
\end{block}
\footnotesize Data: CFRP interstage thermoelastic full-field (DSPSS, 13\,942
nodes) + H3 fairing guided waves.
\end{columns}
\end{frame}

\begin{frame}{Stage 0 --- a 30-line statistic beats eight learned methods}
\begin{columns}[T]
\column{0.58\textwidth}
\includegraphics[width=\textwidth]{../paper_figs/anomaly_zoo_benchmark.pdf}
\column{0.40\textwidth}
Score $s_i = |x_i - \mu_i|/\sigma_i$, with $(\mu_i,\sigma_i)$ per node from
$N{=}2000$ healthy fields. \textbf{Node AUROC (clean):}
\begin{itemize}\small
  \item Mahalanobis / PCA: \textbf{0.999 / 0.995} (2x2 / 4x4)
  \item PaDiM 0.96, PatchCore 0.93, STFPM 0.75, GAE 0.66,
        diffusion 0.50--0.66
\end{itemize}
\begin{alertblock}{SNR limit (right panel)}
\small All methods collapse by $\sigma{=}0.1$: defect amplitude ${<}0.1\sigma$.
Lock-in IR thermography averages $\sim\!10^3$ cycles ($\sigma\!\propto\!1/\sqrt{K}$)
$\Rightarrow$ an SNR \emph{requirement}, not a blocker.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Why the simple method wins here --- and why GNN still matters}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Fixed geometry = template statistics}
Re-inspecting the \emph{same} vehicle: healthy manifold is a point cloud
around one template. Per-node $(\mu_i, \sigma_i)$ is the correct inductive bias.
\end{block}
\column{0.48\textwidth}
\begin{block}{Complementary weaknesses}
\scriptsize
\begin{tabular}{lcc}
\toprule
 & Mahalanobis & GNN \\
\midrule
unknown defect size & \textcolor{okgreen}{open} & \textcolor{alertred}{collapses} \\
noise ($12\%$) & \textcolor{alertred}{dies} & \textcolor{okgreen}{robust} \\
semantic class & --- & \textcolor{okgreen}{only option} \\
new airframe & \textcolor{alertred}{needs refs} & \textcolor{okgreen}{transferable} \\
\bottomrule
\end{tabular}
\end{block}
\end{columns}
\vspace{2mm}
$\Rightarrow$ statistics dominate as a vehicle accumulates flight history;
learned models bootstrap \emph{new} airframes and repairs.
\end{frame}

\begin{frame}{Stage 3 --- phase-field prognosis with true fatigue}
Energy $\;\Psi=\int \big[g(d)\,\psi^+_e
+ \tfrac{G_c}{c_w}(\tfrac{d^2}{\ell}+\ell\,\nabla d\!\cdot\!A\,\nabla d)\big]\,dV$,
with structural tensor $A=I+\beta\,a\!\otimes\!a$ (fiber dir.\ $a$).
\begin{columns}[T]
\column{0.52\textwidth}
\textbf{Model} (12/12 tests green):
\begin{itemize}\small
  \item multi-ply: per-ply $A(\theta_{\text{ply}})$, horizontal bands
  \item interface bands $G_c^{int}/G_c^{ply}{=}0.4$ (DCB/ENF-anchored)
        --- pure-PF surrogate of Paggi--Reinoso CZM
  \item Carrara (2020) fatigue: history $\bar\alpha$ accumulates $\psi^+_e$
        over cycles, $f(\bar\alpha)$ degrades $G_c$ $\Rightarrow$ W\"ohler-like
        growth under \emph{repeated sub-critical} loads
\end{itemize}
\column{0.45\textwidth}
\textbf{Verified physics:}
\begin{itemize}\small
  \item fiber at $30^\circ \to$ crack deflects to $23.3^\circ$
  \item low-$G_c$ interface captures crack (73 vs 0 new cells)
  \item $P(\text{growth})$ monotone in load: $0 \to \tfrac23 \to 1$
  \item FD, plane-strain, $60\times48$ grid, $\sim$15\,s/run
\end{itemize}
\footnotesize\emph{Caveat:} load is a normalised Mode-I peel proxy;
$\bar\alpha_T$ not yet S--N-anchored.
\end{columns}
\end{frame}

\begin{frame}{Flight clearance --- end-to-end on a real test sample}
Stage 0$\rightarrow$3 run: held-out 4x4 field $\rightarrow$ Mahalanobis map
$\rightarrow$ \emph{trained} FMPE posterior $\rightarrow$ fatigue-aware clearance
(20-flight horizon).
\begin{center}
\begin{tabular}{lccccc}
\toprule
load case & load & fatigue & $P(\text{grow})$ & $\mathbb{E}[\text{flights}]$ & decision \\
\midrule
nominal ascent & 0.06 & \textbf{ON}  & 0.00 & \textbf{19.9} & \textcolor{okgreen}{\textbf{OK}} \\
nominal ascent & 0.06 & off & 0.00 & 11.0 & \textcolor{okgreen}{OK} \\
max-Q gust     & 0.10 & \textbf{ON}  & 1.00 & 0.0  & \textcolor{alertred}{\textbf{RETIRE}} \\
max-Q gust     & 0.10 & off & 0.50 & 1.0  & \textcolor{alertred}{RETIRE} \\
\bottomrule
\end{tabular}
\end{center}
\begin{itemize}
  \item fatigue \emph{accumulates across flights} --- $P(\text{survive }20)$ now
        decays even at constant sub-critical load (no i.i.d.\ shortcut)
  \item thresholds from expected cost (vehicle:repair $=100{:}1 \Rightarrow
        \alpha{=}0.02,\ \beta{=}0.48$)
\end{itemize}
\end{frame}

\begin{frame}{Stage 4 --- fleet learning: the SpaceX loop, formalised}
\centering
\includegraphics[width=0.97\textwidth]{../paper_figs/fleet_learning.pdf}\\[1mm]
\begin{itemize}\small
  \item hierarchical Bayes: fleet prior $\varphi \to$ per-vehicle growth rate
        $s_v \to$ inspection posteriors; conjugate Gibbs
  \item (a) $\varphi$ sharpens with flights (SD $0.39\!\to\!0.10$);
        (b) low-data vehicles shrink toward the fleet mean (error ratio 0.84)
  \item (c) \alert{a new vehicle clears in 1.25 inspections with the fleet
        prior vs 2.75 in isolation} --- the fleet-leader effect, quantified
\end{itemize}

\begin{frame}{Outlook: Stage 4--5 and deployment}
\begin{itemize}
  \item \textbf{Stage 4 --- fleet learning:} hierarchical Bayes over tail numbers;
        formalises the SpaceX fleet-leader practice (lead vehicle = likelihood,
        fleet = prior update)
  \item \textbf{Stage 5 --- design feedback:} fleet damage statistics
        $\rightarrow$ layup/reinforcement optimisation (BO)
  \item \textbf{Sim-to-real:} flow-matching bridge ready for first measured
        full-field data (per-sample normalisation needs no healthy reference)
  \item \textbf{Targets:} JAXA reusable-vehicle programmes; journal paper
        (anomaly benchmark + calibrated FMPE + fatigue prognosis)
\end{itemize}
\vspace{2mm}
\begin{block}{One sentence}
SpaceX learned this loop empirically over a decade --- we formalise it so it
works from the first inspection of the first article.
\end{block}
